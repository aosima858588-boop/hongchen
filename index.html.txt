        // 收益计算器逻辑
        document.addEventListener('DOMContentLoaded', function() {
            // 获取DOM元素
            const amountSlider = document.getElementById('amountSlider');
            const amountValue = document.getElementById('amountValue');
            const daysSlider = document.getElementById('daysSlider');
            const daysValue = document.getElementById('daysValue');
            const dailyEarnings = document.getElementById('dailyEarnings');
            const totalEarnings = document.getElementById('totalEarnings');
            const spinButton = document.getElementById('spinButton');
            const wheel = document.getElementById('wheel');
            
            // 格式化数字显示
            function formatNumber(num) {
                return num.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ",");
            }
            
            // 计算阶梯收益率
            function getTieredRate(amount) {
                if (amount < 500) return 0.007; // 0.7%
                if (amount < 2000) return 0.009; // 0.9%
                if (amount < 5000) return 0.010; // 1.0%
                return 0.012; // 1.2%
            }
            
            // 计算收益
            function calculateEarnings() {
                const amount = parseFloat(amountSlider.value);
                const days = parseFloat(daysSlider.value);
                
                // 获取阶梯收益率
                const dailyRate = getTieredRate(amount);
                
                // 计算日均收益
                const dailyEarn = amount * dailyRate;
                // 计算总收益
                const totalEarn = dailyEarn * days;
                
                // 更新显示
                dailyEarnings.innerHTML = `${dailyEarn.toFixed(1)}<span class="result-unit-compact">USDT</span>`;
                totalEarnings.innerHTML = `${totalEarn.toFixed(0)}<span class="result-unit-compact">USDT</span>`;
                
                // 更新数值显示
                amountValue.textContent = formatNumber(amount);
                daysValue.textContent = days;
            }
            
            // 绑定滑块事件
            amountSlider.addEventListener('input', calculateEarnings);
            daysSlider.addEventListener('input', calculateEarnings);
            
            // 初始化计算
            calculateEarnings();
            
            // 抽奖功能
            let isSpinning = false;
            
            spinButton.addEventListener('click', function() {
                if (isSpinning) return;
                
                isSpinning = true;
                spinButton.disabled = true;
                spinButton.innerHTML = '<i class="fas fa-spinner fa-spin"></i> 抽奖中...';
                
                // 生成随机旋转角度
                const extraSpins = 5; // 额外旋转圈数
                const randomAngle = Math.floor(Math.random() * 360);
                const totalRotation = 360 * extraSpins + randomAngle;
                
                // 应用旋转动画
                wheel.style.transition = 'transform 3s cubic-bezier(0.2, 0.8, 0.3, 1)';
                wheel.style.transform = `rotate(${totalRotation}deg)`;
                
                // 显示抽奖结果
                setTimeout(function() {
                    const prizes = [
                        "恭喜！获得 50 USDT 奖励！",
                        "恭喜！获得 H代币空投 奖励！",

                        "恭喜！获得 下次活动优先参与权！",
                        "很遗憾，这次没有中奖，下次再来！"
                    ];
                    
                    const resultIndex = Math.floor(Math.random() * prizes.length);
                    alert(prizes[resultIndex]);
                    
                    // 重置按钮状态
                    spinButton.innerHTML = '<i class="fas fa-gift"></i> 再次抽奖';
                    spinButton.disabled = false;
                    isSpinning = false;
                    
                    // 重置转盘位置（无动画）
                    setTimeout(function() {
                        wheel.style.transition = 'none';
                        wheel.style.transform = 'rotate(0deg)';
                    }, 100);
                }, 3200);
            });
            
            // 添加页面元素进入动画
            const observerOptions = {
                threshold: 0.1,
                rootMargin: '0px 0px -50px 0px'
            };
            
            const observer = new IntersectionObserver(function(entries) {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        entry.target.classList.add('fade-in');
                        observer.unobserve(entry.target);
                    }
                });
